package main

import (
	"encoding/json"
	"errors"
	"io"
	"net/http"
	"os"
	"sync"
	"time"

	"go.bytecodealliance.org/pkg/wasihttp"
	witRuntime "go.bytecodealliance.org/pkg/wit/runtime"
)

// Match auth.go; the production transport itself is copied without changes.
const authJWKSBytes = 64 << 10
const authFetchTimeout = 5 * time.Second

var imports sync.Mutex
var errAuthUnavailable = errors.New("authentication keys unavailable")

func finishImports() { witRuntime.Unpin(); imports.Unlock() }
func main()          {}
func init()          { wasihttp.HandleFunc(serve) }

func fetch(path string) string {
	r, err := http.NewRequest(http.MethodGet, os.Getenv("FIXTURE_URL")+path, nil)
	if err != nil {
		return err.Error()
	}
	r.Header.Set("Authorization", "must-not-be-forwarded")
	response, err := (keyTransport{}).RoundTrip(r)
	if err != nil {
		return err.Error()
	}
	defer response.Body.Close()
	content, err := io.ReadAll(response.Body)
	if err != nil {
		return err.Error()
	}
	return string(content)
}

func serve(w http.ResponseWriter, r *http.Request) {
	if r.URL.Path == "/ready" {
		w.WriteHeader(http.StatusOK)
		return
	}
	start := time.Now()
	result := fetch(r.URL.Path)
	elapsed := time.Since(start).Milliseconds()
	// Exercise cleanup in this same instance, before its resources are destroyed.
	followup := fetch("/success")
	imports.Lock()
	defer finishImports()
	json.NewEncoder(w).Encode(struct {
		Result   string
		Elapsed  int64
		Followup string
	}{result, elapsed, followup})
}
